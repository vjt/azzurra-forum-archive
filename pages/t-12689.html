aggiungo un brevissimo esempio:procedure TForm1.btnConnetti(Sender: TObject);
begin
IdIRC1.Host := 'irc.azzurra.org';
IdIRC1.Port := 6667;
IdIRC1.Username := 'quokka';
IdIRC1.Realname := 'Delphino';
IdIRC1.Nick := 'Delphino';
IdIRC1.Connect(-1);
end;

procedure TForm1.btnDisconnetti(Sender: TObject);
begin
IdIRC1.Disconnect;
end;

gli altri comandi devi darli una volta connesso, analizzando gli eventi OnMessage
