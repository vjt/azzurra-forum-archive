}

int defrag(int ds_main) {

int ris;
indice msg;
int n_mess;
/*creo un file ausilario dove andrò a mettere solo le strutture 'piene' cioè quelle non cancellate*/

int ds_aux = open("auxfile", O_WRONLY|O_TRUNC|O_CREAT, 0666);
if (ris == -1)
Errore_("Errore nella creazione del file\n");

/*con questo ciclo while vado a copiare una per una le strutture 'piene' del file da riordinare nel file ausiliario*/

ris = read(ds_main, &n_mess, 4);
if (ris == -1)
Errore_("Errore nella lettura del file\n");

int n = n_mess -1;

// scrittura nel nuovo file main del numero di messaggi presenti
ris = write(ds_aux, &n, 4);
if (ris == -1)
Errore_("Errore nella scrittura del file\n");

int i = 1;
int trovato = 0:

// copia di tutte le strutture dal main file al file ausiliario
while(i<=n_mess){
i++;
memset(&msg,0,sizeof(indice));
ris = read(ds_main, &msg, sizeof(indice));
if (ris == -1)
Errore_("Errore nella lettura del file\n");
if(msg.mess_id == -1){
trovato = 1;
continue;
}
if(trovato)
msg.mess_id -= 1; // aggiornamento id
ris = write(ds_aux, &msg, sizeof(indice));
if (ris == -1)
Errore_("Errore nella scrittura del file\n");
}

/*adesso cancello il vecchio file main non ordinato */

ris = open(filename, O_TRUNC, 0666);
if (ris == -1)
Errore_("Errore nella modifica del file pointer\n");



// rinomina del file ausiliario in main file
ris = rename("auxfile", mainfile);
if (ris == -1)
Errore_("Errore nella modifica del file pointer\n");



return ds_aux;
}

int elimina (void * pid_client, int j) {
ind_inviato indice_sel; //struttura che contiene l'indice del msg da cancellare
indice msg_info; //struttura che deve contenere info sul messaggio da eliminare
long offset = 4; //serve x trovare la struttura nel file di indici
int ris; //verifica il "ritorno" di alcune funzioni
int ds_main; //descrittore del file di indici
int ds_text; //serve x mettere il nome del file da cancellare
indice auxmsg; //struttura ausiliaria

struct sembuf oper;

//sblocco semaforo j
oper.sem_op = 1;
oper.sem_flg = 0;
oper.sem_num = j;

ris = semop ((int) ds_sem1, &oper, 1);
if (ris == -1)
Errore_("Errore nella chiamata semop sul semaforo 1\n");




int id_coda_server = msgget(KEY_SERVER, IPC_CREAT | 0666); //apre coda del sever
if(id_coda_server == -1) {
Errore_("Errore nell'apertura della coda di msg del server\n");
}

ris = msgrcv(id_coda_server, indice_sel, sizeof(ind_inviato), pid_client, 0);
if (ris == -1)
Errore_("Errore nella chiamata della msgrcv\n");


//ciclo che blocca tutti i semafori in lettura e in scrittura
oper.sem_op = -1;
oper.sem_flg = 0;
int i;
for (i=0; i<num_client; i++) {
oper.sem_num = i;
ris = semop ((int) ds_sem1, &oper, 1);
if (ris == -1)
Errore_("Errore nella chiamata semop sul semaforo 1\n");
}



ds_main = open(mainfile, O_RDWR, 0660);
if (ds_main == -1)
Errore_("Errore nell'apertura del file\n");

/*trova il file da cancellare nel 'file di indici'*/

int size = sizeof(indice);
int pos = indice_sel.n_mess - 1;
offset = offset + (pos * size) + 1; //ora sappiamo dove andare a leggere nel file

ris = lseek(ds_main, offset, 0);//posizioniamo il file pointer sulla struttura giusta
if (ris == -1)
Errore_("Errore nel posizionamento del file pointer\n");

ris = read(ds_main, &msg_info, sizeof(indice));//mettiamo info del file da eliminare in
if (ris == -1) //una struttura
Errore_("Errore nella lettura del file\n");

/*una volta trovato il file giusto, ne identifica il nome e lo cancella*/

ds_text = open(msg_info.file_txt, O_TRUNC, 0660);
if (ris == -1)
Errore_("Errore nell'eliminazione del file\n");

/*ora bisogna rimuovere la struttura dal 'file di indici'...*/


ris = lseek(ds_main, offset, 0); /*questa riposiziona l'indice di lett/scritt*/
if (ris == -1)
Errore_("Errore nella modifica del file pointer\n");


/*...per farlo copio una struttura vuota dello stesso tipo con campo mess_id uguale a
-1 che indica una posizione vuota*/

auxmsg.mess_id = -1;
ris = write(ds_main, &auxmsg, sizeof(indice));
if (ris == -1)
Errore_("Errore nell'aggiornamento del file\n");

ds_main = defrag(ds_main);

close(ds_main);







// sblocco dei semafori
oper.sem_op = 1;
for (i=0; i<num_client; i++) {
oper.sem_num = i;
ris = semop ((int) ds_sem1, &oper, 1);
if (ris == -1)
Errore_("Errore nella chiamata semop sul semaforo 1\n");
}

}

} param;
param par_eli;

ris = creat(mainfile, 0666); //crea il file di indici

Errore_("Errore nella creazione del file di indici\n");

Errore_("Errore nella creazione del semaforo 1\n");

//il semaforo 2 serve per bloccare sovrascritture della struttura da passare alle funzioni

if (ris != 0)
Errore_("Errore nella crezione del thread\n");
}
else if (type == 2) {
ris = pthread_create(&tid, NULL, scrivi, (void*) &richiesta);
if (ris != 0)
Errore_("Errore nella crezione del thread\n");
}
}
