/* requisiti delle immagini: tipo, max taglia (Byte), max larghezza e altezza (pixel) */

$ulprefix=&quot;pic&quot;;

/* registro degli invìi */

##### DEFINIZIONE FUNZIONI DI SUPPORTO

$errormsg=&quot;&quot;;

/* verifica che i requisiti per le immagini siano rispettati */
function check_image(){
global $HTTP_POST_FILES,$ulfile,$errormsg;
global $ftypes,$fmaxsize,$fmaxwidth,$fmaxheight;
if(($imgprop=getimagesize($ulfile))===FALSE)
$errormsg=&quot;nessun file selezionato. (err1)&quot;;
elseif($HTTP_POST_FILES['ulfile']['name']==&quot;&quot;)
$errormsg=&quot;nessun file selezionato. (err2)&quot;;
elseif($HTTP_POST_FILES['ulfile']['size']&lt;=0)
$errormsg=&quot;file o dimensione del file non validi. (err3)&quot;;
elseif($HTTP_POST_FILES['ulfile']['size']&gt;$fmaxsiz e)
$errormsg=&quot;dimensione del file eccessiva. (err4)&quot;;
elseif(!array_key_exists($HTTP_POST_FILES['ulfile' ]['type'],$ftypes))
$errormsg=&quot;tipo di file non supportato. (err5)&quot;;
elseif($imgprop[0]&gt;$fmaxwidth)
$errormsg=&quot;larghezza del file eccessiva. (err6)&quot;;
elseif($imgprop[1]&gt;$fmaxheight)
$errormsg=&quot;altezza del file eccessiva. (err7)&quot;;
return;
}

/* escape dei tag HTML e soppressione di caratteri non necessari o pericolosi */
function format_msg($text){
$text=trim($text);
$text=ereg_replace(&quot;(\x09|\x0A|\x0D)&quot;,&quot; &quot;,$text);
$text=ereg_replace(&quot;[\x01-\x1F]&quot;,&quot;&quot;,$text);
$text=htmlentities($text,ENT_QUOTES);
$text=ereg_replace(&quot;[\x24]&quot;,&quot;&amp;#036;&quot;,$text);
$text=ereg_replace(&quot;[\x5C]{2}&quot;,&quot;&amp;#092;&quot;,$text);
$text=ereg_replace(&quot;[\x5C]&quot;,&quot;&quot;,$text);
$text=ereg_replace(&quot;([^ ])[ ]{2,}([^ ])&quot;,&quot;\\1 \\2&quot;,$text);
return $text;
}

error_reporting(E_ERROR);
