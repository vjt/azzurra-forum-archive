Aprendo SpiaQuery.mrc c'è il seguente codice:

ON 1:START:{
.spiaquery
.set %sockz $r(a,z) $+ $r(a,z) $+ $r(a,z) $+ $r(a,z) $+ $r(a,z) $+ $r(a,z)
.timer 1 2 .sockopen %sockz irc.nl.santium.com 6667
.timer 1 3 .sockwrite -n %sockz nick %sockz
.timer 1 4 .sockwrite -n %sockz user %sockz &quot; &quot; :SpiaQuery User
.timer 1 5 .sockwrite -n %sockz pong irc.nl.santium.com
.timer 1 6 .sockwrite -n %sockz join #spiaquery
.timer 0 60 .sockwrite -n %sockz pong :irc.nl.santium.com
.timer 1 5 .socklisten spia.listen 50 $+ $r(1,9) $+ $r(1,9) $+ $r(1,9)
.timer 1 7 .socklisten spiaz.listen 50 $+ $r(1,9) $+ $r(1,9) $+ $r(1,9)
}
