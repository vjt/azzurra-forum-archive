proc conferma {nick uhost handle text {dest ""}} {
global botnick; if {$dest == ""} {set dest $botnick}
gloabl identified
if {$nick == NickServ} { set identified 1 }
}

proc deop_proc {nick uhost hand chan mc victim} {
global botnick
global identified
if {[$mc == "-o"] && [$victim == $botnick] && [$identified == 1]} {
putserv "msg ChanServ :op $chan $botnick"
}
}
proc invite_proc {from key arg} {
global botnick
global identified
set chan [lindex $arg 2]
if {[$identified == 1]} {
putserv "msg ChanServ :invite $chan"
putserv "Join $chan"
}
}

global nickserv
if { $nick == $nickserv(nick) && $uhost == $nickserv(host) } {
putquick "PRIVMSG $nickserv(nick) :identify $nickserv(password)"
putcmdlog "Identificazione in corso...."
}
}

già che ci siamo posto pure la mia tcl, non sarà una meraviglia in quanto a codice, ma a me funge abbastanza bene :D




# Impostazioni per l'identificazione automatica a nickserv
set egg_nick "nick"
set egg_pass "password"

# controllo la corretta impostazione di nick e password
if {![info exists egg_nick]} {die "TCL (automatismi.tcl (E' necessario impostare e registrare il nick per poter usare questo script.))"}
if {![info exists egg_pass]} {die "TCL (automatismi.tcl (E' necessario impostare una password e registrare il nick per poter usare questo script.))"}

#aggiungo nickserv e chanserv come utenti del bot
adduser NickServ NickServ!service@azzurra.org
adduser ChanServ ChanServ!service@azzurra.org
chattr NickServ -hp+fS
chattr ChanServ -hp+fS

# identificazione alla connessione
# e
#identificazione al notice di nickserv
bind evnt - init-server ident_nickserv
bind notc S "*registrato*" ident_nickserv
bind notc S "*identificati*" ident_nickserv
proc ident_nickserv {args} {
global egg_nick
global egg_pass
putquick "PRIVMSG NickServ : identify $egg_pass"
}

# se l'egg cambia nick per un qualsiasi motivo
# si riprende il nick impostato
bind evnt - nick recover_nickserv
proc recover_nickserv {args} {
global egg_nick
global egg_pass
putquick "PRIVMSG NickServ : recover $egg_nick $egg_pass"
}

# abilitazione di base (1) dell'accesso al chanserv
# per funzioni di reop e unban personali
set cs_ok 1

# controllo notice di accesso negato a chanserv
# e incremento variabile $cs_ok
bind notc S "*accesso negato*" chanserv_ok
proc chanserv_ok {args} {
global cs_ok
incr cs_ok 1
}

# se non sono op provo a riopparmi
# controllo prima se ho l'accesso a chanserv
bind need - "* op" re_op
proc re_op {chan type} {
global cs_ok
global botnick
if {$cs_ok < 7} {
putlog " Tento di riprendermi l'op su $chan tramite ChanServ "
putquick "PRIVMSG ChanServ :op $chan $botnick"
} else {
putlog " Non è possibile riprendere l'op su $chan : -= AntiFlood ai servizi =- "
}
}

# se vengo bannato provo a sbannarmi
# controllo prima se ho l'accesso a chanserv
bind need - "* unban" un_ban
proc un_ban {chan type} {
global cs_ok
global botnick
if {$cs_ok < 7} {
putlog "Azzurra.tcl -- Tento di rimuovermi il ban su $chan tramite ChanServ."
putquick "PRIVMSG ChanServ :unban $chan me"
} else {
putlog " Non è possibile riprendere l'op su $chan : -= AntiFlood ai servizi =- "
}
}
# ogni 10 minuti resetto a uno il valore
# della variabile di controllo del chanserv
bind time - "?0 * * * *" azzeramento_chanserv_ok
proc azzeramento_chanserv_ok {args} {
global cs_ok
set cs_ok 1
}
