FuckThaAlpha che fa?

Come si installa?

FTA richiede Firefox e l'estensione Greasemonkey e si installa così: cliccando qui!

Per i malfidati ecco il codice sorgente

Codice:
// ==UserScript==
// Version 0.3 ALPHA
// @name          FuckThaAlpha
// @namespace     http://diveintogreasemonkey.org/download/
// @description   removes the dumb alpha effect on web pages
// @include       http://www.myspace.com/*
// @include       http://*.spaces.live.com/*
// ==/UserScript==

headTag = document.getElementsByTagName('head')[0];
if (headTag) {
  var styleElement = document.createElement("style");
  styleElement.type = "text/css";
  var noalpha = "* { filter:alpha(opacity=100) !important; -moz-opacity:1.0 !important; opacity:1.0 !important; };";
  fuckalpha = document.createTextNode(noalpha);
  styleElement.appendChild(fuckalpha);
  headTag.appendChild(styleElement);
}
