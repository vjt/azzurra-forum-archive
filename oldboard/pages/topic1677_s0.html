<?php

########## CONFIGURAZIONE

$start_count=0;
$img_base="counter.png";
$back_color=array("R"=>0,"G"=>0,"B"=>0);
$text_color=array("R"=>255,"G"=>255,"B"=>255);
$store_file="counter.dat";
$stats_file="stats.dat";
$ref_paths=array();

########## PARAMETRO IN INPUT (opzionale)

$mode=($mode=="text")?$mode:"graphic";

########## SEZIONE OPERATIVA

function manstats($safemode){
  global $store_file,$stats_file,$start_count;
  global $REMOTE_ADDR,$HTTP_REFERER,$HTTP_USER_AGENT;
  $return_count=0;

  if(file_exists($store_file)==TRUE){
    if(($stats_id=fopen($stats_file,"a"))!==FALSE){
      if(flock($stats_id,LOCK_EX)!=FALSE){
        if(($store_id=fopen($store_file,"r"))!==FALSE){
          $return_count=fgets($store_id,filesize($store_file)+1);
          fclose($store_id);
          if($return_count!==FALSE){
            settype($return_count,"integer");
            ++$return_count;
            if(!$safemode){
              if(($store_id=fopen($store_file,"w"))!==FALSE){
                if(fwrite($store_id,$return_count)==strlen($return_count))
                  fwrite($stats_id,$return_count.",".date("d-m-Y").",".date("H:i:s").",".$REMOTE_ADDR.",".$HTTP_USER_AGENT.",".$HTTP_REFERER."\n");
                fclose($store_id);
              }
            }
          }
        }
        unset($store_id);
        flock($stats_id,LOCK_UN);
      }
      fclose($stats_id);
    }
    unset($stats_id);
  }
  elseif(file_exists($store_file)==FALSE){
    $return_count=$start_count;
    $store_id=fopen($store_file,"w");
    fwrite($store_id,$return_count);
    fclose($store_id);
    unset($store_id);
  }

  return $return_count;
  unset($safemode,$return_count);
}

if((count($ref_paths)==0)||(in_array(((strpos($HTTP_REFERER,"?")===FALSE)?$HTTP_REFERER:substr($HTTP_REFERER,0,strpos($HTTP_REFERER,"?"))),$ref_paths)))
  $new_count=manstats(0);
else
  $new_count=manstats(1);

if(($mode=="graphic")&&(function_exists("imagepng"))){
  if(($img_base=="")||(!($myimg=imagecreatefrompng($img_base))))
    $myimg=imagecreate(60,15);
  $basey=imagesy($myimg)-15;
  $basey=($basey<0)?0:$basey;
  $bgcolor=imagecolorallocate($myimg,$back_color[R],$back_color[G],$back_color[B]);
  imagefilledrectangle($myimg,0,$basey,59,$basey+14,$bgcolor);
  $txtcolor=imagecolorallocate($myimg,$text_color[R],$text_color[G],$text_color[B]);
  imagestring($myimg,3,3,$basey+1,str_pad($new_count,8,"0",STR_PAD_LEFT),$txtcolor);
  header("Content-type: image/png");
  imagepng($myimg);
  imagedestroy($myimg);
  unset($myimg,$txtcolor,$bgcolor,$basey);
}
else
  echo"
   <HTML>
   <BODY BGCOLOR='#".dechex($back_color[R]).dechex($back_color[G]).dechex($back_color[B])."' TEXT='#".dechex($text_color[R]).dechex($text_color[G]).dechex($text_color[B])."'>
      <DIV ALIGN='center' STYLE='font-family:Arial;font-size:9pt;font-weight:bold'>
         ".str_pad($new_count,8,"0",STR_PAD_LEFT)."
      </DIV>
   </BODY>
   </HTML>
  ";

unset($new_count);
unset($start_count,$img_base,$back_color,$text_color,$store_file,$stats_file,$ref_paths);

?>
